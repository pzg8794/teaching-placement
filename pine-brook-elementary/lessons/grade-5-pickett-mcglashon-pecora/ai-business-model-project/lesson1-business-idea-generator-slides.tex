\begin{frame}[plain]
\frametitle{What Is a Business Model?}
\begin{columns}[T,totalwidth=\textwidth]
\column{0.48\textwidth}
\begin{block}{A business model explains}
\begin{itemize}
\large
\item who you help
\item what problem you solve
\item what you offer
\item why people would want it
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Our extra rule}
\Large Your idea must connect to computer science.
\end{block}
\vspace{0.15cm}
\large Apps, data, algorithms, websites, games, robots, privacy, accessibility, or automation can all count.
\end{columns}
\end{frame}

\begin{frame}[plain]
\frametitle{Computer Science Connection Menu}
\begin{columns}[T,totalwidth=\textwidth]
\column{0.48\textwidth}
\begin{itemize}
\large
\item app or website
\item game or simulation
\item data tool
\item algorithm or automation
\end{itemize}
\column{0.48\textwidth}
\begin{itemize}
\large
\item robot, sensor, or 3D design
\item privacy or cybersecurity
\item accessibility technology
\item digital communication
\end{itemize}
\end{columns}
\vspace{0.25cm}
\begin{center}
\Large\bfseries If you cannot explain the CS connection, the idea is not ready yet.
\end{center}
\end{frame}

\begin{frame}[plain]

\begin{block}{Interest}
\Large I like recess and sports.
\end{block}
\begin{block}{Possible business}
\Large RecessBuddy: an app that helps students choose fair teams and game options.

\begin{block}{CS connection}

\begin{frame}[plain]

\begin{frame}[plain]
\frametitle{What Not To Do}
\begin{columns}[T,totalwidth=\textwidth]
\column{0.48\textwidth}
\begin{block}{Do}
\begin{itemize}
\large
\item answer the AI questions
\item compare ideas
\item choose one you understand
\item check the CS connection
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Do Not}
\begin{itemize}
\large
\item share private information
\item pick an idea just because AI said it
\item choose something too huge
\item skip the CS connection
\end{itemize}
\end{block}
\end{columns}
\end{frame}
